search lists matching articles, pages, projects, sources, and storage operations. It does not call the API. execute runs one snippet below as an async function body. Inside a snippet, await blog.request({ method, path, query, body }). It returns { status, body }. Other people's routes, including /auth/me, are refused.

Run every search first, then each execute block as its own execute call, in order. Pass the code exactly, including the console.log lines. Do not wrap it in another function. The last execute must fail. In your final message, quote each successful return value and the refusal. The refusal contains the word refused.

### Execute: drafts

```js
const specs = [
  {
    title: "Harbor Notes",
    markdown: "The 6am ferry leaves before the bakery opens. I counted twelve cormorants on the breakwater and wrote the tide as 1.4 meters, falling.\n\nThis draft is the place I keep the numbers that do not belong in the logbook: who was on the dock, which lamp was out, and whether the green skiff came back.",
    tags: ["harbor", "notes"]
  },
  {
    title: "Night Shift",
    markdown: "The night shift starts when the last bus turns around at the station. The radio sits on the shelf above the kettle, and the kettle takes longer than the announcements.\n\nI keep a list of trains that left late and the names of the people who asked if the rain would stop. It usually does not.",
    tags: ["night", "shift"]
  }
];
const created = [];
for (let i = 0; i < specs.length; i++) {
  const spec = specs[i];
  const response = await blog.request({
    method: "POST",
    path: "/blog/drafts",
    body: { title: spec.title, markdown: spec.markdown, tags: spec.tags }
  });
  if (response.status !== 201) {
    throw new Error("create draft failed " + response.status + " " + JSON.stringify(response.body));
  }
  created.push({
    id: response.body.data.article.id,
    slug: response.body.data.article.slug,
    title: response.body.data.article.draft_title
  });
}
const listed = await blog.request({
  method: "GET",
  path: "/blog/articles",
  query: { status: "drafts", articlesPerPage: 100, page: 1 }
});
if (listed.status !== 200) {
  throw new Error("list drafts failed " + listed.status);
}
console.log("drafts-created", created.length);
return { created: created, listed: listed.status };
```

### Execute: edit

```js
const listed = await blog.request({
  method: "GET",
  path: "/blog/articles",
  query: { status: "drafts", articlesPerPage: 100, page: 1 }
});
let article = null;
const items = listed.body.data.articles;
for (let i = 0; i < items.length; i++) {
  if (items[i].article.draft_title === "Harbor Notes") {
    article = items[i].article;
  }
}
if (!article) {
  throw new Error("Harbor Notes was not listed");
}
const updated = await blog.request({
  method: "POST",
  path: "/blog/articles/" + article.slug + "/update",
  body: {
    title: "Harbor Notes, revised",
    content: "The 6am ferry leaves before the bakery opens. I counted twelve cormorants on the breakwater and wrote the tide as 1.4 meters, falling.\n\nThis draft is the place I keep the numbers that do not belong in the logbook: who was on the dock, which lamp was out, and whether the green skiff came back.\n\nRevised after the first draft. The lamp on the north pier was repaired before noon, so the count of dark lamps is now zero.",
    tags: ["harbor", "revised"],
    image_url: ""
  }
});
if (updated.status !== 200) {
  throw new Error("update failed " + updated.status + " " + JSON.stringify(updated.body));
}
const slug = updated.body.data.article.slug;
const versions = await blog.request({
  method: "GET",
  path: "/blog/articles/" + slug + "/versions"
});
if (versions.status !== 200) {
  throw new Error("versions failed " + versions.status);
}
console.log("draft-revised", slug, versions.body.data.versions.length);
return {
  id: updated.body.data.article.id,
  slug: slug,
  title: updated.body.data.article.draft_title,
  versionCount: versions.body.data.versions.length
};
```

### Execute: page

```js
let created = await blog.request({
  method: "POST",
  path: "/dashboard/pages",
  body: {
    slug: "mcp-demo-desk",
    title: "Desk",
    content: "A short page written through MCP.",
    description: "Demo page for the MCP harness.",
    image_url: "",
    is_published: false
  }
});
let pageId = null;
if (created.status === 201) {
  pageId = created.body.data.id;
} else if (created.status === 409) {
  const listed = await blog.request({
    method: "GET",
    path: "/dashboard/pages",
    query: { page: 1, perPage: 100 }
  });
  const pages = listed.body.data.pages;
  for (let i = 0; i < pages.length; i++) {
    if (pages[i].slug === "mcp-demo-desk") {
      pageId = pages[i].id;
    }
  }
} else {
  throw new Error("page create failed " + created.status + " " + JSON.stringify(created.body));
}
if (!pageId) {
  throw new Error("mcp-demo-desk was not created");
}
const edited = await blog.request({
  method: "PUT",
  path: "/dashboard/pages/" + pageId,
  body: {
    title: "Desk, revised",
    content: "A short page written through MCP. Revised after the first save."
  }
});
if (edited.status !== 200) {
  throw new Error("page update failed " + edited.status + " " + JSON.stringify(edited.body));
}
console.log("page-revised", edited.body.data.slug);
return { id: edited.body.data.id, slug: edited.body.data.slug, title: edited.body.data.title };
```
